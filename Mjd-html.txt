    <script>
        // Main Application Logic
        const codeInput = document.getElementById('codeInput');
        const amountInput = document.getElementById('amountInput');
        const resultValue = document.getElementById('resultValue');
        const toggleBtn = document.getElementById('toggleBtn');
        const toggleText = document.getElementById('toggleText');
        const eyeIcon = document.getElementById('eyeIcon');
        const indicatorDots = document.querySelectorAll('.indicator-dot');
        const keys = document.querySelectorAll('.key');

        let activeField = 'code';
        let isAmountHidden = true;

        function setActiveField(field) {
            activeField = field;
            codeInput.classList.toggle('active', field === 'code');
            amountInput.classList.toggle('active', field === 'amount');
            indicatorDots.forEach(dot => dot.classList.toggle('active', dot.dataset.field === field));
        }

        codeInput.addEventListener('click', () => setActiveField('code'));
        amountInput.addEventListener('click', () => setActiveField('amount'));
        indicatorDots.forEach(dot => dot.addEventListener('click', () => setActiveField(dot.dataset.field)));

        toggleBtn.addEventListener('click', () => {
            isAmountHidden = !isAmountHidden;
            amountInput.classList.toggle('hidden-value', isAmountHidden);
            toggleText.textContent = isAmountHidden ? 'مخفی' : 'نمایش';
            eyeIcon.innerHTML = isAmountHidden 
                ? `<path d="M17.94 17.94A10.07 10.07 0 0 1 12 20c-7 0-11-8-11-8a18.45 18.45 0 0 1 5.06-5.94M9.9 4.24A9.12 9.12 0 0 1 12 4c7 0 11 8 11 8a18.5 18.5 0 0 1-2.16 3.19m-6.72-1.07a3 3 0 1 1-4.24-4.24"></path><line x1="1" y1="1" x2="23" y2="23"></line>`
                : `<path d="M1 12s4-8 11-8 11 8 11 8-4 8-11 8-11-8-11-8z"></path><circle cx="12" cy="12" r="3"></circle>`;
        });

        keys.forEach(key => {
            key.addEventListener('click', () => {
                const keyValue = key.dataset.key;
                const action = key.dataset.action;
                const currentInput = activeField === 'code' ? codeInput : amountInput;
                let value = currentInput.value;

                if (keyValue) {
                    if (keyValue === '.' && value.includes('.')) return;
                    currentInput.value = value + keyValue;
                    calculateResult();
                } else if (action) {
                    switch (action) {
                        case 'delete': currentInput.value = value.slice(0, -1); calculateResult(); break;
                        case 'clear': currentInput.value = ''; calculateResult(); break;
                        case 'clearAll': codeInput.value = ''; amountInput.value = ''; calculateResult(); break;
                        case 'calculate': calculateResult(); resultValue.classList.add('updating'); setTimeout(() => resultValue.classList.remove('updating'), 300); break;
                    }
                }
            });
        });

        function calculateResult() {
            const code = parseFloat(codeInput.value) || 0;
            const amount = parseFloat(amountInput.value) || 0;
            const result = code * amount;
            resultValue.textContent = result.toLocaleString('fa-IR');
            resultValue.classList.add('updating');
            setTimeout(() => resultValue.classList.remove('updating'), 150);
        }

        document.addEventListener('keydown', (e) => {
            if (/^[0-9.]$/.test(e.key)) {
                const currentInput = activeField === 'code' ? codeInput : amountInput;
                if (e.key === '.' && currentInput.value.includes('.')) return;
                currentInput.value += e.key;
                calculateResult();
            }
            if (e.key === 'Backspace') {
                const currentInput = activeField === 'code' ? codeInput : amountInput;
                currentInput.value = currentInput.value.slice(0, -1);
                calculateResult();
            }
            if (e.key === 'Tab') { e.preventDefault(); setActiveField(activeField === 'code' ? 'amount' : 'code'); }
            if (e.key === 'Enter') { calculateResult(); }
        });

        setActiveField('code');

        // PWA Service Worker Registration
        if ('serviceWorker' in navigator) {
            window.addEventListener('load', () => {
                navigator.serviceWorker.register('./sw.js')
                    .then(reg => console.log('Service Worker Registered', reg))
                    .catch(err => console.log('Service Worker Registration Failed', err));
            });
        }
    </script>
